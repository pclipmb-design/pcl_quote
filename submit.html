⚙️ Setup Guide — One-time configuration

Step 1 — Create Google Sheet

  1. 前往 sheets.google.com 建立新試算表
  2. 命名為「PCL Sample Submissions」
  3. 第一列留空（Apps Script 會自動建立標題列）
  4. 複製網址列中的試算表 ID（/d/XXXXXXXX/edit 中間那段）

Step 2 — 部署 Google Apps Script

  1. 在試算表中點選 Extensions → Apps Script
  2. 清除預設程式碼，貼上以下程式碼：
const SHEET_ID = '貼上你的試算表ID';
const LAB_EMAIL = 'pclipmb@gate.sinica.edu.tw';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let ws = ss.getSheetByName('Submissions');
    if (!ws) {
      ws = ss.insertSheet('Submissions');
      ws.appendRow(['Timestamp','Date','Lab/Room','PI','Contact','Email','Phone',
        'Project','Proj No.','Bio Source','Separation','Buffer','PTM',
        'Instrument','Mode','Database','Notes','Services',
        'Sample#','Sample Name','Vol(uL)','Service Items','Core No.','p/inj','MS Note']);
      ws.getRange(1,1,1,25).setFontWeight('bold').setBackground('#1e3a5f').setFontColor('#ffffff');
    }
    const ts = new Date().toISOString();
    data.samples.forEach((s, i) => {
      ws.appendRow([ts, data.date, data.lab, data.pi, data.contact, data.email,
        data.phone, data.project, data.projnum, data.bio, data.sep, data.buffer,
        data.ptm, data.inst, data.mode, data.db, data.notes, data.svcs,
        i+1, s.name, s.vol, s.svc, s.coreNo, s.pinj, s.note]);
    });
    MailApp.sendEmail({
      to: LAB_EMAIL,
      subject: `[PCL Submission] ${data.contact} – ${data.lab} – ${data.date} (${data.samples.length} samples)`,
      htmlBody: buildEmailHtml(data)
    });
    if (data.email) {
      MailApp.sendEmail({
        to: data.email,
        subject: `[PCL] Your submission confirmed – ${data.date}`,
        htmlBody: `
Dear ${data.contact},
Your sample submission has been received by Proteomics Core Lab.
Samples submitted: ${data.samples.length}
Date: ${data.date}
Lab: ${data.lab}
Please bring your samples to A227, Agricultural Technology Building.
We will contact you when processing begins.
Best regards,
Proteomics Core Lab, IPMB/ABRC
`
      });
    }
    return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error',message:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function buildEmailHtml(d) {
  let rows = d.samples.map((s,i) =>
    `${i+1}${s.name}${s.vol}${s.svc}${s.note||''}`
  ).join('');
  return `
New Sample Submission — Proteomics Core Lab

  

  
Date${d.date}

  
Lab / Room${d.lab}

  
Contact${d.contact} — ${d.email} — ${d.phone}

  
Project${d.project} (${d.projnum})

  
Instrument${d.inst} / ${d.mode}

  
Services${d.svcs}

  

  ${rows}

  
#Sample NameVol(μL)ServicesNotes

  
`;
}

function doGet(e) {
  return ContentService.createTextOutput('PCL Submission API is running.').setMimeType(ContentService.MimeType.TEXT);
}
  1. 將第1行的 SHEET_ID 換成你的試算表 ID
  2. 點選 Deploy → New deployment
  3. Type 選 Web app
  4. Execute as: Me  |  Who has access: Anyone
  5. 點 Deploy，複製產生的 Web app URL

Step 3 — 更新此 HTML 檔案

找到檔案最底部 JS 中的這行，替換 URL：

const APPS_SCRIPT_URL = 'YOUR_APPS_SCRIPT_URL';  // ← 換成你的 Web App URL

完成後上傳到 GitHub Pages，即可正式使用。
